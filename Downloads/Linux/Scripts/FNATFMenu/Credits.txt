event ObjectMain
	switch object.state
		case 0
			Credits.Move = 0
		break
		case 1	
			if Credits.Move < 1850
				Credits.Move++
			else
				object.state++
			end if

			if inputpress.buttonB == 1
				object.state++
			end if
		break
		case 2
			foreach (TypeName[Menu Control], arrayPos0, ALL_ENTITIES)
				object[arrayPos0].state = 14
			next	
			object.state = 0
		break
	end switch
end event


event ObjectDraw
	temp0 = 0
	temp1 = 10
	temp1 -= Credits.Move
	while temp0 < 90
		//DrawString(first sprite, x position, y position, space width, character spacing, line drop, menu slot, menu entry)
		DrawString(1, 50, temp1, 4, 0, 1, 5, temp0)
		temp0 ++
		temp1 += 20
	loop
end event
