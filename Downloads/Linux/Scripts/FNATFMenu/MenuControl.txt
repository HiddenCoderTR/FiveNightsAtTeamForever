function MenuControl_DrawText
    LoadSpriteSheet("FNATFMenu/Text.gif")

    CopyPalette(1, 218, 1, 240, 5)
    SetActivePalette(1, 0, 255)

    if MenuControl_ExtrasStrings == true
        temp0 = object.extrasHighlight   
    else
        temp0 = object.highlight
    end if

    if temp0 == arrayPos0
        object.arrowYTarget = object.textY

        if object.colorTimer > 15
            CopyPalette(1, 198, 1, 240, 5)
        else
            CopyPalette(1, 218, 1, 240, 5)
        end if

        DrawSpriteScreenFX(26, FX_INK, object.textX, object.arrowY)        
    end if    
    
    GetTextInfo(temp0, 1, TEXTINFO_TEXTSIZE, object.textLine, 0)
    temp1 = 0
    temp3 = object.textX
    while temp1 < temp0
        GetTextInfo(temp2, 1, TEXTINFO_TEXTDATA, object.textLine, temp1)
        if temp2 > 64
            temp2 -= 65
            GetTableValue(temp6, temp2, MenuControl_HeaderSpace)
            if temp5 == 0 //Y following an A
                if temp2 == 24
                    temp3 -= 4
                end if
            end if
            temp5 = temp2
            DrawSpriteScreenFX(temp2, FX_INK, temp3, object.textY)
            temp3 += temp6
        else
            temp3 += 4
        end if
        temp1++
    loop

    LoadSpriteSheet("FNATFMenu/Menu.gif")
    SetActivePalette(0, 0, 255)
   	CopyPalette(1, 218, 1, 240, 5) 
end function

function MenuControl_MovementFunc
	//SLIDING STUFF
	temp0 = object.movementInputTarget
	temp0 -= object.movementInput
	Abs(temp0)

	temp1 = object.movementInputTarget
	temp1 -= object.movementInput

	//Make sure it's exactly in the spot
	if temp0 < 6
		if object.movementInput > object.movementInputTarget
			object.movementInput--
		else
			if object.movementInput < object.movementInputTarget
				object.movementInput++
			else
				object.movementInput = object.movementInputTarget
			end if
		end if
	else
		temp1 >>= 2
		object.movementInput += temp1
	end if	
end function
