function Options_GrabSaveRAM
	temp0 = object.currentLine
	temp0 += 2

	temp1 = 0
	temp2 = 1000
	temp3 = 0
	while temp1 < 4
		GetTextInfo(temp4, object.propertyValue, TEXTINFO_TEXTDATA, temp0, temp1)
		temp4 -= 48
		temp4 *= temp2	
		temp3 += temp4

		temp2 /= 10
		temp1++
	loop		
	checkResult = temp3	
end function

function Options_GetSelectionCount
	temp0 = object.currentLine
	Options_CurrentSelections = 0
	GetTextInfo(temp1, object.propertyValue, TEXTINFO_ROWCOUNT, 0, 0)
	
	while temp0 < temp1
		GetTextInfo(temp2, object.propertyValue, TEXTINFO_TEXTDATA, temp0, 0)
		if temp2 == TEXTID_MONEYSIGN
			temp0 = temp1
		else
			Options_CurrentSelections++	
		end if
		temp0++
	loop
end function

function Options_MovementFunc
	//SLIDING STUFF
	temp0 = object.movementInputTarget
	temp0 -= object.movementInput
	Abs(temp0)

	temp1 = object.movementInputTarget
	temp1 -= object.movementInput

	//Make sure it's exactly in the spot
	if temp0 < 6
		if object.movementInput > object.movementInputTarget
			object.movementInput--
		else
			if object.movementInput < object.movementInputTarget
				object.movementInput++
			else
				object.movementInput = object.movementInputTarget
			end if
		end if
	else
		temp1 >>= 2
		object.movementInput += temp1
	end if	
end function

function Options_DrawText
	if object.menuLine == MenuControl_OptionsHighlight
		CopyPalette(1, 227, 1, 224, 3)
        if object.textAlign == TEXT_LEFT_ALIGN
            object.highlighterXTarget = object.textX
            object.highlighterYTarget = object.textY
        end if 
	else
		CopyPalette(1, 233, 1, 224, 3)
	end if

	if object.textAlign == TEXT_LEFT_ALIGN
		GetTextInfo(temp0, object.propertyValue, TEXTINFO_TEXTSIZE, object.currentLine, 0)
		temp1 = 0
		temp3 = object.textX
		while temp1 < temp0
			GetTextInfo(temp2, object.propertyValue, TEXTINFO_TEXTDATA, object.currentLine, temp1)
        	if temp2 == TEXTID_NUMSIGN
				temp1 = temp0
			else
                if temp2 >= 192
                    temp2 -= 192
                    GetTableValue(temp6, temp2, charspaceAccented)
                    temp2 += 152

                    if temp2 > 32
                        temp2 -= 33

            GetTextInfo(temp0, object.propertyValue, TEXTINFO_TEXTSIZE, object.currentLine, 0)
            temp1 = 0
            temp3 = object.textX    
			while temp0 > temp1
            	temp0--
				GetTextInfo(temp2, object.propertyValue, TEXTINFO_TEXTDATA, object.currentLine, temp0)

                        temp3 -= 6

private function Options_ChangeLanguage
    switch saved.language
    case 0 // English
        LoadTextFile(2, "EN/DisplayEN.txt")
        LoadTextFile(3, "EN/SoundEN.txt")
        LoadTextFile(4, "EN/GameEN.txt")
        break

    case 1 // Portuguese
        LoadTextFile(2, "PTBR/DisplayPTBR.txt")
        LoadTextFile(3, "PTBR/SoundPTBR.txt")
        LoadTextFile(4, "PTBR/GamePTBR.txt")
        break

    case 2 // Spanish
        LoadTextFile(2, "ES/DisplayES.txt")
        LoadTextFile(3, "ES/SoundES.txt")
        LoadTextFile(4, "ES/GameES.txt")
        break

    case 3 // French
        LoadTextFile(2, "FR/DisplayFR.txt")
        LoadTextFile(3, "FR/SoundFR.txt")
        LoadTextFile(4, "FR/GameFR.txt")
        break
    end switch
end function

event ObjectMain

    object.inkEffect = INK_ALPHA

    switch object.state
    case 0
        object.alpha = 0

        if inputPress.buttonB == 1
            object.state = 3
        end if

        object.state++
        break
    case 1
        object.movementInput = object.alpha
        object.movementInputTarget = 255
        CallFunction(Options_MovementFunc)
        object.alpha = object.movementInput

        if inputPress.buttonB == 1
            object.state = 3
        end if

        if object.alpha == 255
            object.state++
        end if
        break
    case 2
        if inputPress.buttonB == 1
            object.state++
        end if
        break   
    case 3
        object.movementInput = object.alpha
        object.movementInputTarget = 0
        CallFunction(Options_MovementFunc)
        object.alpha = object.movementInput

        if object.alpha == 0
            ResetObjectEntity(object.entityPos, 0, 0, 0, 0)
        end if
        break
    end switch
    if MenuControl_OptionsCur == 0
        temp0 = inputPress.left
        temp0 |= inputPress.right
        if temp0 == 1
            PlaySfx(SfxName[Error], false)
        end if

        switch MenuControl_OptionsHighlight
            case 0 //Display Highlighted
                break
            case 1 // Fullscreen
                temp0 = inputPress.left
                temp0 |= inputPress.right
                if temp0 == 1
                    saved.fullscreen ^= 1
                    CallNativeFunction2(SetWindowFullScreen, saved.fullscreen, 0)
                    CallNativeFunction(ApplyWindowChanges)
                end if
                break

            case 2 // Window Scale		
                if inputPress.left == 1					
                    saved.scale--
                    if saved.scale < 1
                        saved.scale = 4
                    end if
                    CallNativeFunction2(SetWindowScale, saved.scale, 0)
                    CallNativeFunction(ApplyWindowChanges)
                end if
                if inputPress.right == 1
                    saved.scale++
                    if saved.scale > 4
                        saved.scale = 1
                    end if
                    CallNativeFunction2(SetWindowScale, saved.scale, 0)
                    CallNativeFunction(ApplyWindowChanges)
                end if
                break
            case 3 // Graphics Mode
                temp0 = 0
                if inputPress.left == 1
                    temp0 = 1
                    saved.graphics--
                    if saved.graphics < 0
                        saved.graphics = 2
                    end if
                end if
                if inputPress.right == 1
                    temp0 = 1
                    saved.graphics++
                    if saved.graphics > 2
                        saved.graphics = 0
                    end if
                end if
                if temp0 == 1
                    LoadPalette("MainMenu.act", 1, 0, 0, 256)
                    temp2 = 0
                    while temp2 < 256
                        GetTableValue(temp1, temp2, BackupPal)
                        SetPaletteEntry(0, temp2, temp1)
                        temp2++
                    loop   
                    CreateTempObject(TypeName[Palette Police], saved.graphics, 0, 0)
                end if				
                break
            case 4 // Language
                if inputPress.left == 1
                    saved.language--
                    if saved.language < 0
                        saved.language = 3
                    end if
                    CallFunction(Options_ChangeLanguage)
                end if
                if inputPress.right == 1
                    saved.language++
                    if saved.language > 3
                        saved.language = 0
                    end if
                    CallFunction(Options_ChangeLanguage)
                end if				
                break
            case 5 // VSync
                temp0 = inputPress.left
                temp0 |= inputPress.right
                if temp0 == 1
                    saved.vsync ^= 1
                    CallNativeFunction2(SetWindowVSync, saved.vsync, 0)
                end if
                break
            case 6 // Scaling Mode
                if inputPress.left == 1					
                    saved.scalemode--
                    if saved.scalemode < 0
                        saved.scalemode = 3
                    end if
                    CallNativeFunction2(SetWindowScaleMode, saved.scalemode, 0)
                end if
                if inputPress.right == 1
                    saved.scalemode++
                    if saved.scalemode > 3
                        saved.scalemode = 0
                    end if
                    CallNativeFunction2(SetWindowScaleMode, saved.scalemode, 0)
                end if
                break
            case 7 // FPS
                temp0 = inputPress.left
                temp0 |= inputPress.right
                if temp0 == 1
                    if saved.fps == 60
                        saved.fps = 50
                    else
                        saved.fps = 60
                    end if
                    CallNativeFunction2(SetFrameRate, saved.fps, 0)
                end if
                break
            case 8 // Reduce Motion
                temp0 = inputPress.left
                temp0 |= inputPress.right
                if temp0 == 1
                    saved.reduceMotion ^= 1
                end if
                break
        end switch
    else
        object.alpha = 0
        object.state = 0    
	end if
end event


event ObjectDraw

		MenuControl_OptionsLimit = 1
		GetTextInfo(temp1, object.propertyValue, TEXTINFO_ROWCOUNT, 0, 0)
		
		while temp0 < temp1
			GetTextInfo(temp2, object.propertyValue, TEXTINFO_TEXTDATA, temp0, 0)
			if temp2 == TEXTID_MONEYSIGN
				MenuControl_OptionsLimit++
			end if
			temp0++
		loop

        SetActivePalette(1, 0, 255)

        object.textX = screen.xcenter
        object.textX -= 75
        object.textX += MenuControl_OptionDescX
        object.textY = 35
        object.textY -= MenuControl_OptionDescY

        CallFunction(Options_Header)
        
        temp0 = screen.xcenter
        temp0 -= 96
        temp0 += MenuControl_OptionDescX

        temp1 = object.alpha
        temp1 >>= 2
        temp2 = 55
        temp2 -= MenuControl_OptionDescY

        object.movementInput = object.highlighterX
        object.movementInputTarget = object.highlighterXTarget
        CallFunction(Options_MovementFunc)
        object.highlighterX = object.movementInput

        object.movementInput = object.highlighterY
        object.movementInputTarget = object.highlighterYTarget
        CallFunction(Options_MovementFunc)
        object.highlighterY = object.movementInput

        if MenuControl_OptionsHighlight > 0
            temp0 = object.highlighterX
            temp0 -= 12
            temp1 = object.highlighterY
            temp1 -= 3
            temp2 = screen.xcenter
            temp2 += 50
        else    
            temp0 = screen.xcenter
            temp0 -= 113
            temp1 = 65
            temp1 -= MenuControl_OptionDescY     
            temp2 = 0

            object.highlighterX = temp0
            object.highlighterY = temp1
            object.highlighterXTarget = temp0
            object.highlighterYTarget = temp1   
        end if


        DrawRect(temp0, temp1, temp2, 20, color2, color2, color2, object.alpha)

		// Draw the options.
		arrayPos0 = 3
		GetTextInfo(arrayPos1, object.propertyValue, TEXTINFO_ROWCOUNT, 0, 0)
		object.menuLine = 1
        object.marginX = screen.xcenter
        object.marginX -= 100
        object.marginX += MenuControl_OptionDescX
		object.textY = 68
		object.textY -= MenuControl_OptionDescY
		
		while arrayPos0 < arrayPos1
			GetTextInfo(temp0, object.propertyValue, TEXTINFO_TEXTDATA, arrayPos0, 0)
			CheckEqual(temp0, TEXTID_MONEYSIGN)
			temp1 = checkResult
			CheckEqual(arrayPos0, 3)
			temp1 |= checkResult
			if temp1 == 1
				object.currentLine = arrayPos0
				if arrayPos0 != 3
					object.currentLine++
				end if	

                if MenuControl_OptionsHighlight == object.menuLine
                    MenuControl_OptionDescLine = object.currentLine
                    MenuControl_OptionDescLine++
                end if

				object.textX = object.marginX
				object.textAlign = TEXT_LEFT_ALIGN
				CallFunction(Options_DrawText)

				CallFunction(Options_GrabSaveRAM)
				object.currentLine += 3

				switch checkResult
				case 10
					object.currentLine += saved.fullscreen
					break	
				case 11
					switch screen.xsize
					case 320
						break
					case 384
						object.currentLine += 2
						break
					case 426
						object.currentLine++
						break
					default
						object.currentLine += 3
						break
					end switch
					break	
				case 12
					object.currentLine += saved.scale
					object.currentLine--
					break	
				case 13
					object.currentLine += saved.vsync
					break	
				case 14
					object.currentLine += saved.scalemode
					break	
                case 15
                    if saved.fps == 60
                        object.currentLine = object.currentLine
                    else
                	    object.currentLine += 1
                    end if
					break	
				case 16
                    object.currentLine += saved.reduceMotion
                    break
				case 17
					if saved.fps == 60
						object.currentLine++
					end if
					break
				case 200
					arrayPos2 = checkResult
					object.currentLine += saveRAM[arrayPos2]
					break       
				case 201 // WHY DID YOU MAKE ME DO THIS???????
                    object.currentLine += saved.graphics
					break     	
                end switch    

                object.textX = object.marginX
				object.textX += screen.xcenter
				object.textX += 15
				object.textAlign = TEXT_RIGHT_ALIGN			
				CallFunction(Options_DrawText)

				object.textY += 21
                object.marginX -= 7
				object.menuLine++
			end if	
			arrayPos0++
		loop

        SetActivePalette(0, 0, 255)
    end if
end event
