private alias Object.Value0 : Object.Timer    
private alias Object.Value1 : Object.TimelyFellowTimer     
private alias Object.Value3 : Object.TimelyFellowRandom   
private alias Object.Value4 : Object.TimelyFellowAI   
private alias Object.Value5 : Object.ScareTimer    // time i wait before i kill you

private alias Object.Value6 : Object.TimelyFellowRise
private alias Object.Value7 : Object.ShakeTimer
private alias Object.Value8 : Object.HeadShake
private alias Object.Value9 : Object.HeadScale
private alias Object.Value10 : Object.ScareXpos

private alias Object.Value11 : Object.ResumeTimer

private alias Object.value12 : Object.CurtainState
private alias Object.Value13 : Object.MoveTime

private alias Object.Value14 : Object.DoorCount	//each time he has come to your door this night
private alias Object.Value15 : Object.PowerSteal

private alias object.state : TimelyFellow.State

public alias saveRAM[202] : saved.reduceMotion

private alias Object[3].Value4 : Object.UsageBar
private alias Object[4].Value3 : Object.LeftLight
private alias Object[4].Value4 : Object.RightLight

// This object handles the AI, the drawing on the cameras is handled in Camera Drawing
private alias 3 : CAM_PIRATESCOVE_3
private alias 7 : CAM_EASTHALL_7


private alias 15 : TIMELYFELLOW_OFFICEDOOR
private alias 16 : TIMELYFELLOW_WAITTOSCARE
private alias 17 : TIMELYFELLOW_SCARE

private alias Object[4].Value5 : Object.LeftDoorClosed
private alias Object[4].Value6 : Object.RightDoorClosed

event ObjectMain
if Object[SLOT_TITLECARD].Type == TypeName[Blank Object]
	switch TimelyFellow.state
	case CAM_EASTHALL_7
		object[6].alpha = 355
		Object.TimelyFellowTimer++
		if Object.TimelyFellowTimer > 135
			if Object.RightDoorClosed == 1
				//0, 1
				rand(temp2, 2)
				Object.CurtainState = temp2

				OfficePower -= Object.PowerSteal
				Object.DoorCount += 5
				Object.PowerSteal += Object.DoorCount 

				object.state = CAM_PIRATESCOVE_3
				PlaySfx(SfxName[Bang], 0)	
				object.TimelyFellowTimer = 0
			else
				StopMusic()
				PlaySfx(SfxName[Jumpscare], 0)		
				object.state = 17
			end if
		end if
	break
	case TIMELYFELLOW_OFFICEDOOR	//at your door
		if Object.RightDoorClosed == 0	
			//prepare to strike
			Object.ScareTimer++
			if Object.ScareTimer >= 350
				//YOU LOSE
				RDoorLocked = 1
				object.state = TIMELYFELLOW_WAITTOSCARE	//enter offfice
				Object.ScareTimer = 0
			end if
		else
			//aw man
			object.timer++
			if object.timer >= 120
				if Object.RightLight == 1
					StopSfx(SfxName[Hum])
					Object.RightLight = 0
					Object.UsageBar--
				end if

				PlaySfx(SfxName[Block], 0)
				Object.TimelyFellowTimer = 0	
				Object.ScareTimer = 0	
				object[6].alpha = 355
				Object.CurtainState = 0
				TimelyFellow.state = CAM_PIRATESCOVE_3		
				object.timer = 0
			end if				
		end if	
	break	

	case TIMELYFELLOW_WAITTOSCARE
		//just accept your fate man
		Object.ScareTimer++
		temp0 = 500
		rand(temp1, 50)
		temp0 += temp1
		if Object.ScareTimer >= temp0
			StopMusic()
			object.frame = 11
			PlayerJumpscared = 1
			object.state++
			Object.ScareTimer = 0
		end if
	break

	case TIMELYFELLOW_SCARE
		PlayerJumpscared = 1
		if Object.TimelyFellowRise > 60
			Object.TimelyFellowRise -= 16
		else
			Object.TimelyFellowRise = 60
		end if
	break

	default	
		Object.TimelyFellowRise = 100
		//Object.TimelyFellowAI = 20

			if CameraEnabled == 0
				if Object.ResumeTimer > 0
					Object.ResumeTimer--
				else
					Object.TimelyFellowTimer++
					if saved.sonic == 1
						Object.MoveTime = 125
					else
						Object.MoveTime = 500
					end if
				//	rand(temp3, 50)
				//	Object.MoveTime += temp3
					if Object.TimelyFellowTimer >= Object.MoveTime 
						Rand(Object.TimelyFellowRandom, 21)
						if Object.TimelyFellowAI > Object.TimelyFellowRandom 

								if Object[6].value1 == TimelyFellow.state
									object[6].alpha = 355
								end if

								switch TimelyFellow.state
								case CAM_PIRATESCOVE_3	//showstage
									switch Object.CurtainState 
									case 0
										Object.CurtainState = 1
									break
									case 1
										Object.CurtainState = 2
									break
									case 2
										Object.CurtainState = 3
									break	
									case 3
										PlaySfx(SfxName[Timely Run], 0)	
										Object.TimelyFellowTimer = 0
										object.state = CAM_EASTHALL_7
									break							
									end switch	
								break
								end switch	

								if Object[6].value1 == TimelyFellow.state
									object[6].alpha = 355
								end if	
								
						end if
						Object.TimelyFellowTimer = 0	
					end if
				end if	
			else
				rand(Object.ResumeTimer, 100)		
			end if	
		break
	end switch

	if object.state < 18
		if OfficePower == 0
			TimelyFellow.state = 18
		end if
	end if
end if	


	// if inputpress.buttonY == true
	// 	TimelyFellow.state = CAM_EASTHALL_7
	// end if


end event

event ObjectDraw
	switch object.state
		case TIMELYFELLOW_SCARE
			if Object.ShakeTimer == 1
				rand(Object.HeadShake, 8)
			end if	

			object.drawOrder = 5
			if Object.ShakeTimer == 0
				if Object.ScareXpos > screen.xcenter
					Object.ScareXpos -= 24
				else
					Object.ShakeTimer = 1
				end if
			end if	

			temp1 = 0
			temp1 += Object.ScareXpos
			temp1 += Object.HeadShake
			
			DrawSpriteScreenXY(1, temp1, Object.TimelyFellowRise)	//body

			if saved.reduceMotion == 0
				Object.HeadScale += 2
				temp0 = 0
				Sin256(temp0, Object.HeadScale)
				temp0 /= 24
				object.scale += temp0
			end if
			DrawSpriteScreenFX(0, FX_SCALE, temp1, Object.TimelyFellowRise)	//head
		break
	end switch
end event

event ObjectStartup
	LoadSpriteSheet("FNATF/Jumpscares.gif")

	// Place the HUD object into reserved object slot 4 and initialise its values
	Object[13].Type = TypeName[TimelyFellow]
	
	// The object should always be run, regardless if it's within camera bounds or not
	Object[13].Priority = PRIORITY_ACTIVE
	
	Object[13].DrawOrder = 3
	Object[13].ScareXpos = screen.xsize

	Object[13].State = CAM_PIRATESCOVE_3


	SpriteFrame(-103, -102, 206, 204, 664, 1)// 0 head	
	SpriteFrame(-75, 90, 150, 92, 513, 58) //	1 bod		
end event
