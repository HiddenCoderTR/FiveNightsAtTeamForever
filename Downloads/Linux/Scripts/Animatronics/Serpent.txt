// Priority
private alias Object.Value0 : Object.Timer    
private alias Object.Value1 : Object.SerpentTimer     
private alias Object.Value3 : Object.SerpentRandom   
private alias Object.Value4 : Object.SerpentAI   
private alias Object.Value5 : Object.ScareTimer    // time i wait before i kill you

private alias Object.Value6 : Object.SerpentRise
private alias Object.Value7 : Object.ShakeTimer
private alias Object.Value8 : Object.HeadShake
private alias Object.Value9 : Object.HeadScale
private alias Object.Value10 : Object.ScareXpos

private alias Object.Value12 : Object.MoveTime

private alias object.state : Serpent.State

// This object handles the AI, the drawing on the cameras is handled in Camera Drawing
private alias 0 : CAM_SHOWSTAGE_0
private alias 1 : CAM_DININGHALL_1
private alias 2 : CAM_BACKSTAGE_2
private alias 3 : CAM_PIRATESCOVE_3
private alias 4 : CAM_WESTHALL_4
private alias 5 : CAM_CLOSET_5
private alias 6 : CAM_WHALLCORNER_6
private alias 7 : CAM_EASTHALL_7
private alias 8 : CAM_EHALLCORNER_8
private alias 9 : CAM_KITCHEN_9
private alias 10 : CAM_BATHROOMS_10

private alias 15 : SERPENT_OFFICEDOOR
private alias 16 : SERPENT_WAITTOSCARE
private alias 17 : SERPENT_SCARE
private alias 18 : SERPENT_DARKNESS

private alias 21 : SERPENT_SCARE_ALT

public alias saveRAM[202] : saved.reduceMotion

private alias Object[4].Value5 : Object.LeftDoorClosed
private alias Object[4].Value6 : Object.RightDoorClosed

event ObjectMain
if Object[SLOT_TITLECARD].Type == TypeName[Blank Object]
	switch Serpent.state
	case CAM_EHALLCORNER_8	//timer will still go if your looking at him, but will fail if so
		Object.SerpentTimer++
		if Object.SerpentTimer >= 350
			//camera on? dont do anything
			if CameraEnabled == 1
				Object.SerpentTimer = 0	
			else
				//serpent doesnt actually go to your window for now

				// this should be disabling the light
				// if Object[2].Value6 == 1
				// 	StopSfx(SfxName[Hum])
				// 	Object[2].Value6 = 0
				// 	PowerUsage--
				// end if
				PlaySfx(SfxName[Serpent Move], 0)	
				SetSfxAttributes(SfxName[Serpent Move], -1, -100)

				object.alpha = 0
				Serpent.state = SERPENT_OFFICEDOOR	
			end if	
		end if	
	break	
	case SERPENT_OFFICEDOOR	//at your door

		if Object.LeftDoorClosed == 0	
			//prepare to strike
			Object.ScareTimer++
			if Object.ScareTimer >= 300
				object.state = SERPENT_WAITTOSCARE	//enter offfice
				Object.ScareTimer = 0
			end if
		else
			//aw man
			object.timer++
			if object.timer >= 120
				//PlaySfx(SfxName[Block], 0)
				Object.SerpentTimer = 0	
				Object.ScareTimer = 0	
				object[6].alpha = 355
				Serpent.state = CAM_EASTHALL_7		
				object.timer = 0
			end if				
		end if	
	break	

	case SERPENT_WAITTOSCARE
		//YOUR GONNA DIE
		Object.ScareTimer++
		temp0 = 500
		rand(temp1, 50)
		temp0 += temp1
		if Object.ScareTimer >= temp0
			StopMusic()
			PlaySfx(SfxName[Serpent Scare], 0)	
			object.frame = 11
			PlayerJumpscared = 1
			object.state++
			Object.ScareTimer = 0
		end if
	break

	case SERPENT_SCARE
		PlayerJumpscared = 1
		if Object.SerpentRise > 80
			Object.SerpentRise -= 16
		else
			Object.SerpentRise = 80
		end if
	break

	case SERPENT_DARKNESS
		Object[3].alpha = 0
		Object.SerpentTimer++
		if Object.SerpentTimer > 512
			PlayMusic(3)
			object.state++
			Object.SerpentTimer = 0
		end if
	break
	// playing the song
	case 19
		Object.SerpentTimer++
		temp0 = 750
		rand(temp1, 500)
		temp0 += temp1
		if Object.SerpentTimer >= temp0
			StopMusic()
			PlaySfx(SfxName[Serpent Move], 0)	
			object.drawOrder = 4
			object.state = 20
		end if
	break
	//time before death
	case 20
		//YOUR GONNA DIE
		Object.ScareTimer++
		temp0 = 500
		rand(temp1, 100)
		temp0 += temp1
		if Object.ScareTimer >= temp0
			StopMusic()
			PlaySfx(SfxName[Serpent Scare], 0)	
			object.frame = 11
			Object.SerpentRise = 200
			PlayerJumpscared = 1
			object.state++
			Object.ScareTimer = 0
		end if
	break

	case SERPENT_SCARE_ALT
		PlayerJumpscared = 1
		if Object.SerpentRise > 80
			Object.SerpentRise -= 16
		else
			Object.SerpentRise = 80
		end if
	break

	
	default	
		Object.SerpentRise = 200
		//Object.SerpentAI = 20


		//if you arent watching him
		

		if cameraenabled == 1
			CheckNotEqual(Object[6].value1, Serpent.State)
			temp0 = checkResult		
		end if
		if temp0 == 1
			Object.SerpentTimer++
			if Object.SerpentTimer >= 350
				//if AI is greater then a random number 0 to 19 (techincally 20 numbers)
				// move
				if saved.sonic == 1
					Object.MoveTime = 5
				else
					Object.MoveTime = 20
				end if
				Rand(Object.SerpentRandom, Object.MoveTime)
				if Object.SerpentAI > Object.SerpentRandom 
					//static!
					if Object[6].value1 == Serpent.State
						object[6].alpha = 355
					end if

					PlaySfx(SfxName[Serpent Move], 0)	
					//these align with the cameras he goes to
					//state 10 is cam 10
					//15+ is at the office watch out
					switch Serpent.state
					case CAM_SHOWSTAGE_0
						Serpent.state = CAM_DININGHALL_1
					break
					case CAM_DININGHALL_1
						Serpent.state = CAM_BATHROOMS_10
					break		
					case CAM_BATHROOMS_10
						Serpent.state = CAM_KITCHEN_9
					break	
					case CAM_KITCHEN_9
						Serpent.state = CAM_EASTHALL_7
					break								
					case CAM_EASTHALL_7
						Serpent.state = CAM_EHALLCORNER_8
					break			
					end switch	
	
					if Object[6].value1 == Serpent.State
						object[6].alpha = 355
					end if

				end if
				Object.SerpentTimer = 0		
			end if
		end if
		break
	end switch

	if object.state < 18
		if OfficePower == 0
			Serpent.state = 18
			Object.SerpentTimer = 0	
			Object.ScareTimer = 0
		end if
	end if
end if	
end event

event ObjectDraw
	switch object.state
		case SERPENT_SCARE
		case SERPENT_SCARE_ALT
			//slides in from the left
			//stops at the screen center

			if Object.ShakeTimer == 1
				rand(Object.HeadShake, 8)
			end if	

			object.drawOrder = 5
			if Object.ShakeTimer == 0
				temp0 = screen.xcenter
				if Object.ScareXpos < temp0
					Object.ScareXpos += 24
				else
					Object.ShakeTimer = 1
				end if
			end if	

			temp1 = 0
			temp1 += Object.ScareXpos
			temp1 += Object.HeadShake

			DrawSpriteScreenXY(1, temp1, Object.SerpentRise)	//body

			if saved.reduceMotion == 0
				Object.HeadScale += 2
				temp0 = 0
				Sin256(temp0, Object.HeadScale)
				temp0 /= 32
				object.scale += temp0
			end if
			DrawSpriteScreenFX(0, FX_SCALE, temp1, Object.SerpentRise)	//head
		break
	end switch
end event

event ObjectStartup
	LoadSpriteSheet("FNATF/Jumpscares.gif")

	// Place the HUD object into reserved object slot 4 and initialise its values
	Object[10].Type = TypeName[Serpent]
	
	// The object should always be run, regardless if it's within camera bounds or not
	Object[10].Priority = PRIORITY_ACTIVE
	
	Object[10].DrawOrder = 3


	SpriteFrame(-88, -141, 176, 281, 1, 1)// 0 head	
	SpriteFrame(-76, 79, 152, 157, 178, 1) //	1 bod		
end event
