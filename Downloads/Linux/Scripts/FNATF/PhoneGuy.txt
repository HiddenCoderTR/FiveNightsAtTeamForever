event ObjectMain
	if Object[SLOT_TITLECARD].Type == TypeName[Blank Object]
		if stage.actnum <= 5
			object.phoneDial = true

			if object.phoneDial == true
				object.phoneTimer++
				if object.phoneTimer == 100
					PlaySfx(SfxName[Phone Dial], false)	
				end if

				if object.phoneTimer == 480
					temp0 = stage.listPos
					temp0 += 82
					PlayVoice(temp0, false)
					Object.phoneLine++
					
					object.phoneDial = 2
					stage.timeEnabled = true
				end if	
			end if	

			if object.phoneTimer > 480

					object.phoneLineTimer++

				switch stage.actnum
					case 1
						GetTableValue(temp0, object.phoneLineTime, PhoneGuy_N1Times)
					break
					case 2
						GetTableValue(temp0, object.phoneLineTime, PhoneGuy_N2Times)
					break
					case 3
						GetTableValue(temp0, object.phoneLineTime, PhoneGuy_N3Times)
					break
					case 4
						GetTableValue(temp0, object.phoneLineTime, PhoneGuy_N4Times)
					break
					case 5
						GetTableValue(temp0, object.phoneLineTime, PhoneGuy_N5Times)
					break
				end switch
	
				if Object.phoneLineTimer >= temp0
					object.phoneLineTime++
					Object.phoneLine++
				end if
			end if

			switch stage.actnum
				case 1
					if Object.phoneTimer > 11000
						temp0 = stage.listPos
						temp0 += 82
						StopVoice(temp0)
						object.type = TypeName[Blank Object]
					end if
				break	
				case 2
					if Object.phoneTimer > 3600
						temp0 = stage.listPos
						temp0 += 82
						StopVoice(temp0)
						object.type = TypeName[Blank Object]
					end if
				break	
				case 3
					if Object.phoneTimer > 3200
						temp0 = stage.listPos
						temp0 += 82
						StopVoice(temp0)
						object.type = TypeName[Blank Object]
					end if
				break		
				case 4
					if Object.phoneTimer > 2900
						temp0 = stage.listPos
						temp0 += 82
						StopVoice(temp0)
						object.type = TypeName[Blank Object]
					end if
				break		
				case 5
					if Object.phoneTimer > 2500
						temp0 = stage.listPos
						temp0 += 82
						StopVoice(temp0)
						object.type = TypeName[Blank Object]
					end if
				break												
			end switch

			if PlayerJumpscared == 1
				temp0 = stage.listPos
				temp0 += 82
				StopVoice(temp0)
				object.type = TypeName[Blank Object]
			end if

			temp0 = 0
			temp0 += 8
			temp1 = 0
			temp1 += 80
			CheckTouchRect(temp0, 40, temp1, 70)
			if CheckResult > -1
				temp0 = stage.listPos
				temp0 += 82
				StopVoice(temp0)			
				object.type = TypeName[Blank Object]
			end if
		end if
	end if
end event
