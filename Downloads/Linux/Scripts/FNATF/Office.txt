function Night_AI
	switch stage.actnum
	case 1
		if stage.minutes < 2
			Object.SerpentAI  = -1
			Object.SoinkAI = -1
			Object.ShrineAI  = -1
			Object.TimelyFellowAI = -1
		else
			Object.SerpentAI  = 0
			Object.SoinkAI = 3
			Object.ShrineAI  = 3
			Object.TimelyFellowAI = 0
		end if
	break
	case 2
		Object.SerpentAI  = 0
		Object.SoinkAI = 5
		Object.ShrineAI  = 4
		Object.TimelyFellowAI = 3
	break
	case 3
		if stage.minutes < 1
			Object.SerpentAI  = 1
		else
			Object.SerpentAI  = 2
		end if
		Object.SoinkAI = 7
		Object.ShrineAI  = 6
		Object.TimelyFellowAI = 5				
	break
	case 4
		Object.SerpentAI  = 2
		Object.SoinkAI = 9
		Object.ShrineAI  = 9
		Object.TimelyFellowAI = 7
	break	
	case 5
		Object.SerpentAI  = 4
		Object.SoinkAI = 10
		Object.ShrineAI  = 10
		Object.TimelyFellowAI = 9				
	break		
	case 6
		Object.SerpentAI  = 6
		Object.SoinkAI = 10
		Object.ShrineAI  = 14
		Object.TimelyFellowAI = 16	
	break		
	case 7
		Object.SerpentAI  = custom.serpent
		Object.SoinkAI = custom.soink
		Object.ShrineAI  = custom.shrine
		Object.TimelyFellowAI = custom.timely	
	break		
	end switch	

	if saved.dog == 1
		if stage.actnum <= 6
			temp0 = 2
			Object.SerpentAI  *= temp0
			Object.SoinkAI *= temp0
			Object.ShrineAI  *= temp0
			Object.TimelyFellowAI *= temp0		
		end if
	end if
end function

// doors and lights functionality
function Office_Controls	
	//if they arent dead and havent won yet
	CheckEqual(PlayerJumpscared, 0)
	temp0 = checkresult
	CheckEqual(NightWon, 0)
	temp0 &= checkresult
	if temp0 == 1
		if inputDown.left == 1
			if Object.ScrollPos < LEFTLIMIT
					Object.ScrollPos += 10
			end if	
		end if

				Object.RightLimit = 537
		Object.RightLimit -= screen.xsize
		not(Object.RightLimit)

		if inputDown.right == 1
			if Object.ScrollPos > Object.RightLimit
				Object.ScrollPos -= 10
			end if
		end if

		// facing the left
		if Object.ScrollPos >= 15
			if inputPress.buttonA == 1
				//if the door isnt locked
				if LDoorLocked == 0
					Object.LeftDoorClosed ^= 1
					PlaySfx(SfxName[Door], 0)
				else
					PlaySfx(SfxName[Error], 0)	
				end if	


				if Object.LeftDoorClosed == 1
					Object.UsageBar += 1
				else
					Object.UsageBar -= 1
				end if
			
			end if

			if saved.lights == 0
				if inputpress.buttonC == 1
					Object.LeftLight ^= 1
					if LWindowScare == 1
						PlaySfx(SfxName[DoorScare], 0)
						LWindowScare = 0
					end if
					if Object.LeftLight == 1
						Object.UsageBar += 1
					else
						Object.UsageBar -= 1
					end if
				end if

				if Object.LeftLight == 1
					if Object.RightLight == 1
						Object.RightLight = 0
						Object.UsageBar -= 1
					end if
				end if	
				if Object.RightLight == 1
					if Object.LeftLight == 1
						Object.LeftLight = 0
						Object.UsageBar -= 1
					end if
				end if	
			end if
	

		// facing the right	
		else
			if inputpress.buttonA == 1
				//if the door isnt locked
				if RDoorLocked == 0
					Object.RightDoorClosed ^= 1
					PlaySfx(SfxName[Door], 0)
				else
					PlaySfx(SfxName[Error], 0)	
				end if	


				if Object.RightDoorClosed == 1
					Object.UsageBar += 1
				else
					Object.UsageBar -= 1
				end if				
			end if

			if saved.lights == 0
				if inputpress.buttonC == 1
					if RWindowScare == 1
						PlaySfx(SfxName[DoorScare], 0)
						RWindowScare = 0
					end if
					Object.RightLight ^= 1

					if Object.RightLight == 1
						Object.UsageBar += 1
					else
						Object.UsageBar -= 1
					end if					
				end if		

				if Object.RightLight == 1
					if Object.LeftLight == 1
						Object.LeftLight = 0
						Object.UsageBar -= 1
					end if
				end if	
				if Object.LeftLight == 1
					if Object.RightLight == 1
						Object.RightLight = 0
						Object.UsageBar -= 1
					end if
				end if
			end if			

		end if

		CheckEqual(Object.LeftLight, 1)
		temp0 = checkresult
		CheckEqual(Object.RightLight, 1)
		temp0 |= checkresult		
		if temp0 == 1
			Object.SoundTimer++
			if Object.SoundTimer >= 0
				PlaySfx(SfxName[Hum], 0)
				Object.SoundTimer = -12
			end if	
		else
			StopSfx(SfxName[Hum])
			Object.SoundTimer = 0
		end if


		if inputPress.buttonZ == 1
			PlaySfx(SfxName[Trigger], 0)
		end if
	end if
end function

function Office_LeftSide
	if object.LeftLight == 1
		temp0 = Object.ScrollPos
		temp0 += 77
		DrawSpriteScreenXY(14, temp0, screen.ycenter)	//door
	end if

	if object.LeftLight == 1
		CheckEqual(Shrine.State, SHRINE_OFFICEDOOR)
		temp1 = checkresult
		CheckEqual(Shrine.State, SHRINE_WAITTOSCARE)
		temp1 |= checkresult		
		if temp1 == 1
			temp0 = Object.ScrollPos
			temp0 += 84
			DrawSpriteScreenXY(19, temp0, screen.ycenter)	//door	
		end if
	end if

	if Object.LeftDoorClosed == 1	//open
		// go down
		if Object.LDoorSlide < 295
			Object.LDoorSlide += 18
		else
			Object.LDoorSlide = 295
		end if
	else
		// go up
		if Object.LDoorSlide > 0
			Object.LDoorSlide -= 18
		else
			Object.LDoorSlide = 0
		end if	
	end if	
	temp0 = Object.ScrollPos
	temp0 += 66
	DrawSpriteScreenXY(13, temp0, Object.LDoorSlide)	//door
end function

event ObjectMain

	CallFunction(Night_AI)

	// if theres no title card object
	if Object[SLOT_TITLECARD].Type == TypeName[Blank Object]
		// if you arent dead, then you can pause
		// no rage quitting sorry
		if PlayerJumpscared == 0
			if OfficePower > 0
				if Object[9].Type == TypeName[Blank Object]

					if useMouse == 1
						temp0 = 0
						temp0 += screen.xsize
						temp0 -= 132
						temp1 = 0
						temp1 += screen.xsize
						temp1 -= 87
						CheckTouchRect(temp0, 0, temp1, 36)
						if CheckResult > -1
							inputpress.start = 1
						end if
					end if

					if inputpress.Start == true
						// Disable start for this frame, this'll help with "double-pausing" cases
						inputpress.Start = false
					
						// Give different pause menus depending on if the player was brought here from the Dev Menu
						// (...but why?)
				
						// Pause the stage, this'll pause all the stage's objects too
						Stage.State = STAGE_FROZEN
						
						// and pause the stage's music, as well
						PauseMusic()
						// Create the Pause Menu and set it up as needed
						Object[9].Type = TypeName[Pause Menu]
						Object[9].State = 0
						Object[9].DrawOrder = 6
						Object[9].Priority = 2
					end if
				end if
			end if
		end if	

		// controls being door controls, and touch controls
		if OfficePower > 0
			if CameraEnabled == 0
				CallFunction(Office_MobileControls)
				CallFunction(Office_Controls)
			end if

			//fan animation
			Object.FanTimer++
			if saved.reduceMotion == 0
				if Object.FanTimer == 3
					Object.FanFrame++
					Object.FanTimer = 0
				end if
			else
				if Object.FanTimer == 9
					Object.FanFrame++
					Object.FanTimer = 0
				end if
			end if
			if Object.FanFrame > 17
				Object.FanFrame = 16
			end if
			Object.FanSound++
			if Object.FanSound >= 0
				PlaySfx(SfxName[Fan], 0)
				Object.FanSound = -572
			end if

			//power stuff

			Object.PowerTimer++
			
			if saved.tails == 1
				Object.PowerDrainRate = 450
			else
				Object.PowerDrainRate = 300
			end if

			temp0 = Object.UsageBar
			temp0 *= 50
			Object.PowerDrainRate -= temp0

			if Object.PowerTimer >= Object.PowerDrainRate
				OfficePower--
				Object.PowerTimer = 0
			end if
		else
			StopMusic()
			if Object.SoundTrigger == 0
				PlaySfx(SfxName[Power Out], 0)
				Object.SoundTrigger = 1
			end if
			StopSfx(SfxName[Hum])
			StopSfx(SfxName[Fan])
			Object.LeftDoorClosed = 0
			Object.RightDoorClosed = 0
			Object.RightLight = 0
			Object.LeftLight = 0
		end if

		//cant move when the camera is up
		CheckEqual(CameraEnabled, 0)
		temp0 = checkresult
		CheckEqual(PlayerJumpscared, 0)
		temp0 &= checkresult
		if temp0 == 1

			if UseMouse == 1
				// scroll left
				if Object.ScrollPos < LEFTLIMIT
					if touchscreen[0].xpos < 170
						Object.ScrollPos += 3
					end if
				end if	
				// fast if on the other side
				if Object.ScrollPos < LEFTLIMIT
					if touchscreen[0].xpos < 120
						Object.ScrollPos += 5
					end if
				end if
				
				Object.RightLimit = 537
				Object.RightLimit -= screen.xsize
				not(Object.RightLimit)
				
			
				temp0 = screen.xcenter
				temp0 += 50
				if Object.ScrollPos > Object.RightLimit
					if touchscreen[0].xpos > temp0
						Object.ScrollPos -= 3
					end if
				end if
				temp0 = screen.xcenter
				temp0 += 90
				if Object.ScrollPos > Object.RightLimit
					if touchscreen[0].xpos > temp0
						Object.ScrollPos -= 5
					end if
				end if
			end if	
		end if
	end if

	// if inputPress.buttonY == true
	// 	OfficePower = 0
	// end if
end event
