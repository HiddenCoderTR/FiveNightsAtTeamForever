event ObjectDraw
	if camera.state == CAMERA_MAIN
		DrawRect(0, 0, Screen.XSize, Screen.YSize, 0, 0, 0, 255)
		DrawSpriteScreenXY(Object.CameraState, screen.xcenter, screen.ycenter)


		if object.camerastate == Shrine.State
			temp0 = screen.xcenter
			temp1 = screen.ycenter
			switch Shrine.State
				case CAM_SHOWSTAGE_0
					temp0 += 128
					temp1 -= 8
					DrawSpriteScreenXY(23, temp0, temp1)
				break
				case CAM_DININGHALL_1
					temp0 -= 125
					temp1 -= 3
					DrawSpriteScreenXY(24, temp0, temp1)
				break
				case CAM_BATHROOMS_10
					temp0 += 8
					temp1 -= 28
					DrawSpriteScreenXY(25, temp0, temp1)
				break		
				//nothing happens in the kitchen
				case CAM_EASTHALL_7
					temp0 -= 72
					temp1 += 64
					DrawSpriteScreenXY(26, temp0, temp1)
				break					
				case CAM_EHALLCORNER_8
					temp0 -= 64
					temp1 += 32
					DrawSpriteScreenXY(27, temp0, temp1)
				break
			end switch
		end if		

		if object.camerastate == Serpent.State
			temp0 = screen.xcenter
			temp1 = screen.ycenter
			switch Serpent.State
				case CAM_SHOWSTAGE_0
					temp0 -= 0
					DrawSpriteScreenXY(12, temp0, temp1)
				break
				case CAM_DININGHALL_1
					temp0 += 42
					temp1 -= 48
					DrawSpriteScreenXY(13, temp0, temp1)
				break
				case CAM_BATHROOMS_10
					temp0 -= 70
					temp1 -= 28
					DrawSpriteScreenXY(14, temp0, temp1)
				break		
				// while the serpent goes to the kitchen, they dont draw anyhting cuzzz yknow
				case CAM_EASTHALL_7
					temp0 += 30
					temp1 -= 28
					DrawSpriteScreenXY(15, temp0, temp1)
				break					
				case CAM_EHALLCORNER_8
					temp0 += 42
					temp1 += 18
					DrawSpriteScreenXY(16, temp0, temp1)
				break	
			end switch
		end if

		if object.camerastate == Soink.State
			temp0 = screen.xcenter
			temp1 = screen.ycenter
			switch Soink.State
				case CAM_SHOWSTAGE_0
					temp0 -= 128
					temp1 += 32
					DrawSpriteScreenXY(17, temp0, temp1)
				break
				case CAM_DININGHALL_1
					temp0 += 42
					temp1 += 52
					DrawSpriteScreenXY(18, temp0, temp1)
				break
				case CAM_BACKSTAGE_2
					temp0 += 22
					temp1 -= 28
					DrawSpriteScreenXY(19, temp0, temp1)
				break		
				case CAM_WESTHALL_4
					temp0 -= 40
					temp1 -= 28
					DrawSpriteScreenXY(20, temp0, temp1)
				break					
				case CAM_CLOSET_5
					temp0 -= 48
					temp1 -= 8
					DrawSpriteScreenXY(21, temp0, temp1)
				break
				case CAM_WHALLCORNER_6
					temp0 += 42
					temp1 += 18
					DrawSpriteScreenXY(22, temp0, temp1)
				break						
			end switch
		end if

		if object.camerastate == TimelyFellow.State
			temp0 = screen.xcenter
			temp1 = screen.ycenter	

			switch TimelyFellow.State
				case CAM_PIRATESCOVE_3
					switch Object.CurtainState
						case 0
							temp0 -= 0
							temp1 -= 12
							DrawSpriteScreenXY(29, temp0, temp1)
						break
						case 1
							temp0 -= 0
							temp1 -= 12
							DrawSpriteScreenXY(30, temp0, temp1)
						break	
						case 2
							temp0 -= 0
							temp1 -= 12
							DrawSpriteScreenXY(31, temp0, temp1)

							temp0 += 16
							temp1 += 32
							DrawSpriteScreenXY(32, temp0, temp1)
						break		
						case 3
							temp0 -= 0
							temp1 -= 12
							DrawSpriteScreenXY(31, temp0, temp1)

							temp0 += 16
							temp1 += 52
							DrawSpriteScreenXY(33, temp0, temp1)
						break			
						default
							temp0 -= 0
							temp1 -= 12
							DrawSpriteScreenXY(31, temp0, temp1)
						break																			
					end switch
				break
			end switch

		end if
	end if
end event
