public alias saveRAM[204] : saved.officeMusic

event ObjectMain
	switch object.state
		case 0
			StopMusic()
			object.alpha = 256
			object.timer++
			if object.timer > 32
				object.state++
				PlaySfx(SfxName[Night Start], 0)
				object.timer = 0
			end if
		break
		case 1
			if object.alpha > 0
				object.alpha -= 8
			end if

			if object.alpha <= 0
				object.timer++
				if object.timer > 200
					object.state++
					object.timer = 0
				end if
			end if
		break
		case 2
			object.alpha += 8
			if object.alpha > 255

				object.timer++
				if object.timer > 100
					if saved.officeMusic == 1
						SetMusicTrack("Office.ogg", 0, 1)
						PlayMusic(0)
					end if
					object.type = TypeName[Blank Object]
					object.timer = 0
				end if
				
			end if
		break
	end switch

	if object.state < 2
		if inputpress[1].start == true
			object.timer = 100
			object.state = 2
		end if
	end if

end event

event ObjectDraw
	DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)

	object.inkeffect = INK_ALPHA
	temp0 = screen.xcenter
	temp0 -= 24
	DrawSpriteScreenXY(10, temp0, screen.ycenter)
	temp0 += 96
	DrawNumbers(0, temp0, screen.ycenter, stage.actnum, 2, 29, 0)
	DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, object.alpha)	

end event

			break
		case 6
			SetPresenceDetails("Night 6")
			break

			SetPresenceDetails("Custom Night")

	end switch
