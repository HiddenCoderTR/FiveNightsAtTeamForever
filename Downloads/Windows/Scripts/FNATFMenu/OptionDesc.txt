event ObjectDraw
    if object.propertyValue == 0
        temp0 = inputPress.left
        temp0 |= inputPress.right
        temp0 |= inputPress.up
        temp0 |= inputPress.down
        if temp0 == true
            foreach(TypeName[Option Desc], arrayPos0, ALL_ENTITIES)
                if object[arrayPos0].propertyValue > 0
                    ResetObjectEntity(arrayPos0, TypeName[Blank Object], 0, 0, 0)
                end if
            next
        end if

        if object.displayDesc == false
            object.descYTarget = 0
            MenuControl_OptionDescY = 25
            MenuControl_OptionDescX = 10
        else
            object.descYTarget = 40
            MenuControl_OptionDescY = 0
            MenuControl_OptionDescX = 0
        end if
        
        if inputPress.buttonY == true
            object.displayDesc ^= 1
        end if

        if inputPress.buttonB == true
            if object.displayDesc == false
                object.displayDesc = true
            end if
        end if

        object.movementInput = object.descY
        object.movementInputTarget = object.descYTarget
        CallFunction(MenuControl_MovementFunc)
        object.descY = object.movementInput

        object.textFile = MenuControl_OptionsCur
        object.textFile += 2

        // option zero override

        //dont make it harder then it has to be
        if MenuControl_OptionsHighlight == 0
            MenuControl_OptionDescLine = 2
        end if
        

        object.wordBorder = screen.xsize
        object.wordBorder -= 16

        object.textY = screen.ysize
        object.textY -= 30
        object.textY += object.descY

        temp0 = object.textY
        temp0 -= 3
	    DrawRect(0, temp0, screen.xsize, 90, color3, color1, color1, 255)

        object.currentWord = 1

        GetTextInfo(temp0, object.textFile, TEXTINFO_TEXTSIZE, MenuControl_OptionDescLine, 0)
        temp1 = 0
        temp3 = 5
        while temp1 < temp0
            GetTextInfo(temp2, object.textFile, TEXTINFO_TEXTDATA, MenuControl_OptionDescLine, temp1)
            if temp2 == 64
                temp3 = 5
                object.currentRow++
                object.currentWord++
                object.textY += 8
            else		
                if temp2 == 32
                    object.currentWord++
                    temp3 += 8
                else	
                    if temp2 >= 192
                        temp2 += 100       		
                    end if            

                    arrayPos0 = 300
                    arrayPos0 += temp1
                    ResetObjectEntity(arrayPos0, TypeName[Option Desc], object.currentWord, 0, 0)
                    object[arrayPos0].ixpos = temp3
                    object[arrayPos0].iypos = object.textY
                    object[arrayPos0].drawOrder = 6
                    object[arrayPos0].frame = temp2
                    object[arrayPos0].priority = PRIORITY_ACTIVE

                    temp3 += 8   
                end if  
            end if 

            if temp3 >= object.wordBorder		
                temp7 = 0		
                object.currentRow++
                object.textY += 8

                temp6 = 5					
                foreach(TypeName[Option Desc], arrayPos0, ALL_ENTITIES)
                    if object[arrayPos0].propertyValue == object.currentWord	
                        if object[arrayPos0].iypos < object.textY

                            object[arrayPos0].ixpos = temp6	
                            object[arrayPos0].iypos += 8	
                                
                            temp7 += 8
                            temp6 += 8																																	
                        end if
                    end if
                next

                temp3 = 5
                temp3 += temp7
            end if 
            temp1++
        loop
	else
		DrawSpriteScreenXY(object.frame, object.ixpos, object.iypos)
	end if			   
end event
