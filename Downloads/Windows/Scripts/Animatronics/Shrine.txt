// This object handles the AI, the drawing on the cameras is handled in Camera Drawing
private alias 0 : CAM_SHOWSTAGE_0
private alias 1 : CAM_DININGHALL_1
private alias 2 : CAM_BACKSTAGE_2
private alias 3 : CAM_PIRATESCOVE_3
private alias 4 : CAM_WESTHALL_4
private alias 5 : CAM_CLOSET_5
private alias 6 : CAM_WHALLCORNER_6
private alias 7 : CAM_EASTHALL_7
private alias 8 : CAM_EHALLCORNER_8
private alias 9 : CAM_KITCHEN_9
private alias 10 : CAM_BATHROOMS_10

private alias 15 : SHRINE_OFFICEDOOR
private alias 16 : SHRINE_WAITTOSCARE
private alias 17 : SHRINE_SCARE

private alias Object[4].Value5 : Object.LeftDoorClosed
private alias Object[4].Value6 : Object.RightDoorClosed

event ObjectMain
if Object[SLOT_TITLECARD].Type == TypeName[Blank Object]
	switch Shrine.state
	case SHRINE_OFFICEDOOR	//at your door
		if Object.LeftDoorClosed == 0	
			//prepare to strike
			Object.ScareTimer++
			if Object.ScareTimer >= 400
				//YOU LOSE
				LDoorLocked = 1
				object.state = SHRINE_WAITTOSCARE	//enter offfice
				Object.ScareTimer = 0
			end if
		else
			Object.ShrineTimer++

			if saved.sonic == 1
				Object.MoveTime = 125
			else
				Object.MoveTime = 500
			end if
			rand(temp3, 100)
			Object.MoveTime += temp3

			if Object.ShrineTimer >= Object.MoveTime
				Rand(Object.ShrineRandom, 20)
				if Object.ShrineAI > Object.ShrineRandom 

					if Object.LeftLight == 1
						StopSfx(SfxName[Hum])
						Object.LeftLight = 0
						Object.UsageBar--
					end if

					if LWindowScare == 1
						LWindowScare = 0
					end if

					//PlaySfx(SfxName[Block], 0)
					Object.ShrineTimer = 0	
					Object.ScareTimer = 0	
					object[6].alpha = 355
					Shrine.state = CAM_DININGHALL_1		
					object.timer = 0
				end if				
				Object.ShrineTimer = 0
			end if
		end if	
	break	

	case SHRINE_WAITTOSCARE
		//just accept your fate man
		Object.ScareTimer++
		temp0 = 500
		rand(temp1, 50)
		temp0 += temp1
		if Object.ScareTimer >= temp0
			StopMusic()
			PlaySfx(SfxName[Jumpscare], 0)	
			object.frame = 11
			PlayerJumpscared = 1
			object.state++
			Object.ScareTimer = 0
		end if
	break

	case SHRINE_SCARE
		PlayerJumpscared = 1
		if Object.ShrineRise > 100
			Object.ShrineRise -= 16
		else
			Object.ShrineRise = 100
		end if
	break

	default	
		Object.ShrineRise = 100
		//Object.ShrineAI = 20


		Object.ShrineTimer++

		Object.MoveTime = 500
		rand(temp3, 100)
		Object.MoveTime += temp3

		if Object.ShrineTimer >= Object.MoveTime
			//if AI is greater then a random number 0 to 19 (techincally 20 numbers)
			// move
			Rand(Object.ShrineRandom, 20)

			//0, 1
			rand(Object.RandomLocation1, 3)

			if Object.ShrineAI > Object.ShrineRandom 

				if Object[6].value1 == Shrine.State
					object[6].alpha = 355
				end if

				PlaySfx(SfxName[Shrine Slide], 0)	

				//these align with the cameras he goes to
				//state 10 is cam 10
				//15+ is at the office watch out
				switch Shrine.state
				case CAM_SHOWSTAGE_0
					Shrine.state = CAM_DININGHALL_1
				break
				case CAM_DININGHALL_1
					switch Object.RandomLocation1
					case 1	
						object.state = CAM_BATHROOMS_10
					break
					case 2	
						PlaySfx(SfxName[Shrine Kitchen], 1)
						object.state = CAM_KITCHEN_9
					break						
					default
						object.state = CAM_EASTHALL_7
					break					
					end switch
				break		
				case CAM_BATHROOMS_10
					switch Object.RandomLocation1
					case 1	
						object.state = CAM_DININGHALL_1
					break
					case 2		
						PlaySfx(SfxName[Shrine Kitchen], 1)
						object.state = CAM_KITCHEN_9
					break						
					default
						object.state = CAM_EASTHALL_7
					break					
					end switch
				break
				case CAM_KITCHEN_9
					switch Object.RandomLocation1
					case 1	
					case 2	
						StopSfx(SfxName[Shrine Kitchen])
						object.state = CAM_DININGHALL_1
					break						
					default
						StopSfx(SfxName[Shrine Kitchen])
						object.state = CAM_EHALLCORNER_8
					break					
					end switch
				break	
				case CAM_EASTHALL_7
					Shrine.state = CAM_EHALLCORNER_8
				break								
				case CAM_EHALLCORNER_8
					LWindowScare = 1

					if Object.LeftLight == 1
						StopSfx(SfxName[Hum])
						Object.LeftLight = 0
						Object.UsageBar--
					end if
					object.alpha = 0
					Shrine.state = 15
				break			
				end switch	

				if Object[6].value1 == Shrine.State
					object[6].alpha = 355
				end if

			end if
			Object.ShrineTimer = 0		
		end if
		break
	end switch

	if object.state < 18
		if OfficePower == 0
			Shrine.state = 18
		end if
	end if
end if	

	// if inputpress.buttonY == 1
	// 	shrine.state = SHRINE_WAITTOSCARE	
	// end if

end event

event ObjectDraw
	switch object.state
		case SHRINE_SCARE
			if Object.ShakeTimer == 1
				rand(Object.HeadShake, 8)
			end if	

			object.drawOrder = 5
			if Object.ShakeTimer == 0
				if Object.ScareXpos < screen.xcenter
					Object.ScareXpos += 24
				else
					Object.ShakeTimer = 1
				end if
			end if	

			temp1 = 0
			temp1 += Object.ScareXpos
			temp1 += Object.HeadShake

			if saved.reduceMotion == 0
				Object.HeadScale += 2
				temp0 = 0
				Sin256(temp0, Object.HeadScale)
				temp0 /= 16
				object.scale += temp0
			end if
			DrawSpriteScreenFX(0, FX_SCALE, temp1, Object.ShrineRise)	//head
		break
	end switch
end event

	LoadSpriteSheet("FNATF/Jumpscares.gif")

	// Place the HUD object into reserved object slot 4 and initialise its values
	Object[12].Type = TypeName[Shrine]
	
	// The object should always be run, regardless if it's within camera bounds or not
	Object[12].Priority = PRIORITY_ACTIVE
	
	Object[12].DrawOrder = 3
	Object[12].ScareXpos = 0


	SpriteFrame(-213, -79, 426, 240, 178, 159)// 0 head	
end event
