event ObjectMain
	switch object.State
	case CAMERA_WAITACTIVATE
		//title card should be gone
		if Object[SLOT_TITLECARD].Type == TypeName[Blank Object]
			
			if UseMouse == 1
				CheckTouchRect(0, 208, screen.xsize, 240)
				if CheckResult > -1
					Object.TouchTimer = 1
					inputpress.buttonB = 1
				end if		
			end if	

			//if you dont do anything for bit then   your gonna get it
			Camera.Timer++
			if Camera.Timer >= 8000
				PlaySfx(SfxName[Timely Run], 0)	
				Object.TimelyFellowTimer = 0
				TimelyFellow.State = CAM_EASTHALL_7
				Camera.Timer = 0
			end if

			//if the plalyer isnt dead or if they havent won yet
			CheckEqual(PlayerJumpscared, 0)
			temp0 = checkResult
			CheckEqual(NightWon, 0)
			temp0 &= checkResult			
			if temp0 == 1
				if OfficePower > 0
					CheckEqual(inputPress.buttonB, 1)
					temp1 = checkResult					
					if temp1 == 1
						PlaySfx(SfxName[Camera], 0)	
						object.animation = 1
						Camera.Frame = 11
						Object.UsageBar += 1
						Camera.Timer = 0
						object.state++
					end if
				end if		
			end if


		end if
	break
	case CAMERA_FLIPUP
		CameraEnabled = 1
		if Camera.Frame == 16
			object.animation = 0
			object.state++
		end if
	break
	case CAMERA_MAIN
		if UseMouse == 1
			CallFunction(Camera_TouchControls)
		end if

		foreach (TypeName[Office], arrayPos0, ALL_ENTITIES)
			CheckEqual(Object[arrayPos0].Value3, 1)
			temp1 = checkResult
			CheckEqual(Object[arrayPos0].Value4, 1)
			temp1 |= checkResult
			if temp1 == 1
				foreach (TypeName[HUD], arrayPos1, ALL_ENTITIES)
					Object[arrayPos1].Value4 -= 1
				next
				StopSfx(SfxName[Hum])
				Object[arrayPos0].Value9 = 0
				Object[arrayPos0].Value3 = 0
				Object[arrayPos0].Value4 = 0
			end if
		next

		if UseMouse == 1
			CheckTouchRect(0, 208, screen.xsize, 240)
			if CheckResult > -1
				Object.TouchTimer = 1
				inputpress.buttonB = 1
			end if		
		end if	

		CheckEqual(PlayerJumpscared, 1)
		temp1 = checkResult
		CheckEqual(inputPress.buttonB, 1)
		temp1 |= checkResult		
		CheckEqual(OfficePower, 0)
		temp1 |= checkResult	
		CheckEqual(ForcePull, 1)
		temp1 |= checkResult	
		if temp1 == 1
			Camera.Frame = 16
			PlaySfx(SfxName[Camera], 0)	
			object.animation = 2
			CameraEnabled = 0
			Object.UsageBar -= 1
			object.state = CAMERA_EXIT
		end if

		
	break
	case CAMERA_EXIT
		if Camera.frame == 11
			ForcePull = 0
			object.animation = 0
			object.state = 0
		end if
	break		
	end switch

	switch object.animation
	case 1
		Object.AnimationTimer += 128
		
		if Object.AnimationTimer > 239
			Object.AnimationTimer -= 240
				
			Camera.Frame++
			if Camera.Frame > 16
				Camera.Frame = 16
			end if
		end if
	break

	//flip down
	case 2
		Object.AnimationTimer += 128
		
		if Object.AnimationTimer > 239
			Object.AnimationTimer -= 240
			
			Camera.Frame--
			if Camera.Frame < 11
				Camera.Frame = 11
			end if
		end if
	break	
	end switch
end event
