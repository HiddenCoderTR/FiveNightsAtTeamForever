private alias object.value0 : object.red
private alias object.value1 : object.blu
private alias object.value2 : object.grn
private alias object.value3 : object.process

public alias saveRAM[294] : saved.graphics

private alias 0 : GRAPHICS_MOBILE         
private alias 1 : GRAPHICS_EMULATED        
private alias 2 : GRAPHICS_HARDWARE

// Priority Aliases
private alias 0 : PRIORITY_BOUNDS
private alias 1 : PRIORITY_ACTIVE
private alias 2 : PRIORITY_ALWAYS

// Values
public value PalettePolice_StoredColor = 0;

// ========================
// Function Declarations
// ========================

reserve function PalettePolice_ColorProcessor
reserve function PalettePolice_HardwareColors


// ========================
// Function Definitions
// ========================

function PalettePolice_ColorProcessor
	switch saved.graphics
	case 0 // Mobile Conversion
		// not used currently... this is the mobile remaster we're talking about...
		//PalettePolice_StoredColor &= 0xE0E0E0
		break

	case 1 // Emulated Conversion
		object.red = PalettePolice_StoredColor
		object.red &= 0xFF0000
		object.red >>= 16
		object.red *= 9
		object.red >>= 3
		if object.red > 255
			object.red = 255
		end if
		object.red <<= 16
					
		object.grn = PalettePolice_StoredColor
		object.grn &= 0x00FF00
		object.grn >>= 8
		object.grn *= 9
		object.grn >>= 3
		if object.grn > 255
			object.grn = 255
		end if
		object.grn <<= 8
					
		object.blu = PalettePolice_StoredColor
		object.blu &= 0x0000FF
		object.blu *= 9
		object.blu >>= 3
		if object.blu > 255
			object.blu = 255
		end if
		
		PalettePolice_StoredColor = object.red
		PalettePolice_StoredColor += object.blu
		PalettePolice_StoredColor += object.grn
		break
		
	case 2 // Hardware Conversion
		object.process = PalettePolice_StoredColor
		object.process &= 0xE00000
		object.process >>= 21
		CallFunction(PalettePolice_HardwareColors)
		object.red = object.process
		object.red <<= 16				
		
		object.process = PalettePolice_StoredColor
		object.process &= 0x00E000
		object.process >>= 13
		CallFunction(PalettePolice_HardwareColors)
		object.grn = object.process
		object.grn <<= 8
		
		object.process = PalettePolice_StoredColor
		object.process &= 0xE0
		object.process >>= 5
		CallFunction(PalettePolice_HardwareColors)
		object.blu = object.process				
		
		PalettePolice_StoredColor = object.red
		PalettePolice_StoredColor += object.blu
		PalettePolice_StoredColor += object.grn
		break
	end switch
end function


function PalettePolice_HardwareColors
	switch object.process
	case 0
		object.process = 0
		break

	case 1
		object.process = 52
		break

	case 2
		object.process = 87
		break

	case 3
		object.process = 116
		break

	case 4
		object.process = 144
		break

	case 5
		object.process = 172
		break

	case 6
		object.process = 206
		break

	case 7
		object.process = 255
		break

	end switch
end function


// ========================
// Events
// ========================

event ObjectMain
	temp0 = 0
	while temp0 < 8
		temp1 = 0
		while temp1 < 256
			GetPaletteEntry(temp0, temp1, PalettePolice_StoredColor)			
			CallFunction(PalettePolice_ColorProcessor)			
			SetPaletteEntry(temp0, temp1, PalettePolice_StoredColor)
			temp1++
		loop
		temp0++
	loop	
		
	// Some stuff drawn by the game like Rectangles and Fades require RGB values to be inputted
	// Since graphics mode exists, we use variables so that way the colors will still be adjusted
	switch saved.graphics
	case 0
		color1 = 32
		color2 = 64
		color3 = 96
		color4 = 128
		color5 = 160
		color6 = 192
		color7 = 224
		break

	case 1
		color1 = 36
		color2 = 72
		color3 = 108
		color4 = 144
		color5 = 180
		color6 = 216
		color7 = 252
		break

	case 2
		color1 = 52
		color2 = 87
		color3 = 116
		color4 = 144
		color5 = 172
		color6 = 206
		color7 = 255
		break			
	end switch

	// Sadly this object can't exist...
	object.type = TypeName[Blank Object]
end event
