event ObjectMain
	switch object.state	
		// fade in
		case 0
			object.timer = 255
			object.state++
			SetClassicFadeIn(0, 0, 0, object.timer) 
		break
		case 1
			if object.timer > 0
				object.timer -= 8
			else
				PlayMusic(0)
				object.state++
			end if
			SetClassicFadeIn(0, 0, 0, object.timer) 
		break	
		case 2
			CallFunction(CN_StupidTouchRects)

			if inputpress.start == true
				PlaySfx(SfxName[Camera], false)
				object.state++
			end if

			if inputpress.buttonB == true
				PlaySfx(SfxName[Camera], false)
				object.state = 4
			end if			
		break	
		case 3
			object.timer += 16
			music.volume -= 8
			if object.timer > 384
				WriteSaveRAM()
				stage.listPos = 6
				stage.activeList = REGULAR_STAGE
				LoadStage()				
			end if	
			SetClassicFadeOut(0, 0, 0, object.timer) 			
		break
		case 4
			object.timer += 16
			music.volume -= 8
			if object.timer > 640
				stage.listPos = 0
				stage.activeList = PRESENTATION_STAGE
				SkipTFScreen = true
				LoadStage()				
			end if	
			SetClassicFadeOut(0, 0, 0, object.timer) 			
		break		
	end switch
end event
